{{- $raw := .Site.Params.ai_search | default dict -}} {{- $enabled := false -}} {{- $provider := "" -}} {{- $websiteID := "" -}} {{- $sourceGroups := dict -}} {{- if reflect.IsMap $raw -}} {{- $enabled = index $raw "enabled" | default false -}} {{- if ne (printf "%T" $enabled) "bool" -}} {{- errorf "params.ai_search.enabled must be a boolean" -}} {{- end -}} {{- $provider = index $raw "provider" | default "" -}} {{- $websiteID = index $raw "website_id" | default "" -}} {{- $sourceGroups = index $raw "source_groups" -}} {{- else -}} {{- errorf "params.ai_search must be a map" -}} {{- end -}} {{- /* This site selects an exact version before exposing the language groups. An unconfigured version keeps a local notice, never a latest fallback. */ -}} {{- $version := .Site.Params.version | default "latest" -}} {{- $versionGroups := dict -}} {{- $configured := false -}} {{- if reflect.IsMap $sourceGroups -}} {{- $configured = isset $sourceGroups $version -}} {{- $versionGroups = index $sourceGroups $version | default dict -}} {{- end -}} {{- $enGroup := "" -}} {{- $cnGroup := "" -}} {{- if reflect.IsMap $versionGroups -}} {{- $enGroup = index $versionGroups "en" | default "" -}} {{- $cnGroup = index $versionGroups "cn" | default "" -}} {{- end -}} {{- if $enabled -}} {{- if not (reflect.IsMap $sourceGroups) }}{{ errorf "params.ai_search.source_groups must be a map when AI search is enabled" }}{{ end -}} {{- if ne $provider "kapa" }}{{ errorf "params.ai_search.provider must be kapa when AI search is enabled" }}{{ end -}} {{- if not $websiteID }}{{ errorf "params.ai_search.website_id is required when AI search is enabled" }}{{ end -}} {{- if and (or $configured (eq $version "latest")) (or (not $enGroup) (not $cnGroup)) -}} {{- errorf "params.ai_search.source_groups.%s.en and .cn are required for configured AI versions" $version -}} {{- end -}} {{- end -}} {{- return (dict "enabled" $enabled "provider" $provider "websiteID" $websiteID "version" $version "sourceGroups" (dict "en" $enGroup "cn" $cnGroup) ) -}}

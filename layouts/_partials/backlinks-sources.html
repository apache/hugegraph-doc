{{- /* HugeGraph exposes backlinks only for latest documentation. Historical builds and non-doc sections retain their original, uncluttered output. */ -}} {{- $page := . -}} {{- $enabled := and (eq ($page.Site.Params.version | default "latest") "latest") (eq $page.Section "docs") -}} {{- if isset $page.Params "backlinks" -}} {{- $enabled = partial "validate.html" (dict "value" (index $page.Params "backlinks") "kind" "bool" "fallback" $enabled "key" "front matter backlinks" "where" $page.Path) -}} {{- end -}} {{- $sources := slice -}} {{- if $enabled -}} {{- $backlinkIndex := partialCached "backlinks-index.html" $page.Site $page.Site.Language.Lang -}} {{- range sort (index $backlinkIndex $page.Path | default (slice)) "Path" -}} {{- $sources = $sources | append . -}} {{- end -}} {{- end -}} {{- return $sources -}}
